# 配置参数
FORMAT = pyaudio.paInt16  # 音频格式

# 全局变量控制录音状态

def record_audio():
    """
    实时录制音频并保存为临时文件
    """
    audio = pyaudio.PyAudio()
    # 打开音频流
    stream = audio.open(format=FORMAT,
        channels=CHANNELS,
        rate=RATE,
        input=True,
        frames_per_buffer=CHUNK)
   
    print("开始实时语音转文字（按Ctrl+C停止）...\n")
    frames = []
   
    # 录制指定时长的音频
    for _ in range(0, int(RATE / CHUNK * RECORD_SECONDS)):
        data = stream.read(CHUNK)
        frames.append(data)
        if kwa['end']:
            break

    kwa['end'] = True
   
    # 保存临时音频文件
    wf = wave.open(TEMP_AUDIO_FILE, 'wb')
    wf.setnchannels(CHANNELS)
    wf.setsampwidth(audio.get_sample_size(FORMAT))
    wf.setframerate(RATE)
    wf.writeframes(b''.join(frames))
    wf.close()
   
    # 停止录音
    stream.stop_stream()
    stream.close()
    audio.terminate()
    wf.close()

def transcribe_audio():
    """
    实时转录音频文件
    """
    # 等待音频文件生成
    time.sleep(0.1)
    if os.path.exists(TEMP_AUDIO_FILE):
        try:
            # 使用Whisper转录
            result = model.transcribe(TEMP_AUDIO_FILE, language="zh", fp16=False)
            text = result["text"].strip()
           
            # 输出转录结果
            print(text, end='')
           
            transmit_thread = threading.Thread(target=transmit_data_to_AI, args=(text, ), daemon=True)
            transmit_thread.start()
        except Exception as e:
            print(f"转录出错: {e}")

def keyboard_listener(event):
    if event.name == 'b':
        kwa['end'] = False
        speech_to_text_thread = threading.Thread(target=record_audio, daemon=True)
        speech_to_text_thread.start()
    elif event.name == 'e':
        kwa['end'] = True

def main():
    keyboard.on_press(keyboard_listener)
    keyboard.wait('esc')

if __name__ == "__main__":
    main()
